import { afterEach, beforeEach, describe, expect, it, vi } from "vitest";

const effects = vi.hoisted(() => ({ cleanup: undefined as undefined | (() => void) }));
vi.mock("react", () => ({
  useEffect: (setup: () => (() => void)) => { effects.cleanup = setup(); },
}));
import { useStaffLoginRipples } from "../client/src/hooks/use-staff-login-ripples";

class Surface extends EventTarget {
  children: Surface[] = [];
  parent: Surface | null = null;
  className = "";
  clientLeft = 1;
  clientTop = 1;
  style: Record<string, unknown> = {
    setProperty: (key: string, value: string) => { this.style[key] = value; },
  };
  setAttribute = vi.fn();
  getBoundingClientRect() { return { left: 100, top: 50, width: 516, height: 503 }; }
  appendChild(child: Surface) { child.parent = this; this.children.push(child); }
  remove() {
    if (this.parent) this.parent.children = this.parent.children.filter(child => child !== this);
    this.parent = null;
  }
}

function pointer(surface: Surface, type: string, x = 160, y = 130, time = 0, pointerType = "mouse") {
  const event = new Event(type, { cancelable: true });
  Object.defineProperties(event, {
    clientX: { value: x }, clientY: { value: y },
    timeStamp: { value: time }, pointerType: { value: pointerType },
  });
  surface.dispatchEvent(event);
  return event;
}

describe("staff login card water interactions", () => {
  let face: Surface;
  let preference: EventTarget & { matches: boolean };
  let busy: { current: boolean };
  beforeEach(() => {
    face = new Surface();
    preference = Object.assign(new EventTarget(), { matches: false });
    busy = { current: false };
    vi.stubGlobal("window", { matchMedia: () => preference });
    vi.stubGlobal("document", { createElement: () => new Surface() });
  });
  afterEach(() => { effects.cleanup?.(); vi.unstubAllGlobals(); });
  const attach = () => useStaffLoginRipples(
    { current: face as unknown as HTMLDivElement }, false, { current: false }, busy,
  );

  it("centers rings on the actual pointer relative to the inner border", () => {
    attach();
    pointer(face, "pointerenter");
    expect(face.children).toHaveLength(1);
    expect(face.children[0].style.left).toBe("59px");
    expect(face.children[0].style.top).toBe("79px");
    expect(face.children[0].children).toHaveLength(3);
  });
  it("throttles hover movement by both distance and time", () => {
    attach();
    pointer(face, "pointerenter");
    pointer(face, "pointermove", 165, 130, 200);
    pointer(face, "pointermove", 230, 130, 100);
    expect(face.children).toHaveLength(1);
    pointer(face, "pointermove", 230, 130, 200);
    expect(face.children).toHaveLength(2);
  });
  it("allows repeated presses and touch without intercepting controls", () => {
    attach();
    for (let index = 0; index < 7; index++) {
      expect(pointer(face, "pointerdown", 160, 130, index, "touch").defaultPrevented).toBe(false);
    }
    expect(face.children).toHaveLength(4);
    expect(face.children.every(child => child.className.includes("--press"))).toBe(true);
  });
  it("cleans a finished ripple and all effects on unmount", () => {
    attach();
    pointer(face, "pointerdown");
    const end = Object.assign(new Event("animationend"), { animationName: "staff-login-ripple-lifetime" });
    face.children[0].dispatchEvent(end);
    expect(face.children).toHaveLength(0);
    pointer(face, "pointerenter");
    effects.cleanup?.();
    pointer(face, "pointerdown");
    expect(face.children).toHaveLength(0);
  });
  it("clears and suppresses motion immediately when reduced motion is enabled", () => {
    attach();
    pointer(face, "pointerenter");
    preference.dispatchEvent(Object.assign(new Event("change"), { matches: true }));
    pointer(face, "pointerdown");
    expect(face.children).toHaveLength(0);
  });
  it("does not generate hover ripples for touch or while authentication is busy", () => {
    attach();
    pointer(face, "pointerenter", 160, 130, 0, "touch");
    busy.current = true;
    pointer(face, "pointerdown");
    expect(face.children).toHaveLength(0);
  });
});
